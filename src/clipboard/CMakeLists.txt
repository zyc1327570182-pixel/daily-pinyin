cmake_minimum_required(VERSION 3.16)
project(daily_pinyin_clipboard LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(PkgConfig REQUIRED)
pkg_check_modules(FCITX REQUIRED IMPORTED_TARGET Fcitx5Module)
add_library(clipboard MODULE clipboard.cpp)
target_compile_definitions(clipboard PRIVATE ENABLE_X11 FCITX_GETTEXT_DOMAIN="fcitx5")
target_include_directories(clipboard PRIVATE /usr/include/Fcitx5/Module/fcitx-module/xcb)
target_link_libraries(clipboard PRIVATE PkgConfig::FCITX)
target_compile_options(clipboard PRIVATE "-ffile-prefix-map=${CMAKE_CURRENT_SOURCE_DIR}=.")
